**Goal:** 선언(map/steps)이 검사 가능한 SQL로 컴파일되어 DuckDB에서 벡터화 실행되는 변환 엔진.

**Architecture:** 각 step → CTE 하나. `gladius compile`이 생성 SQL을 그대로 노출(투명성). 엔진은 in-process DuckDB — `read_parquet` glob으로 읽고 `COPY (…) TO`로 쓴다. Arrow 경유라 Pugio 산출물과 zero-copy로 맞물린다.

**Tech Stack:** duckdb(in-process), pyarrow, hypothesis(SQL 파싱 퍼징), pytest golden test

## DuckDB 활용 방침 (엔진은 빌린다 — 우리가 쓰는 기능 목록)

| 기능 | 용도 |
|---|---|
| `read_parquet('dir/**/*.parquet')` | 입력 — Pugio unit 파일들을 글롭으로 한 번에. 스키마 union은 `union_by_name=true` |
| `COPY (query) TO 'out' (FORMAT PARQUET, COMPRESSION ZSTD)` | 출력 — 중간 materialization 없이 스트리밍 쓰기 |
| `SELECT * EXCLUDE (...)` / `* REPLACE (...)` | rename/cast CTE를 짧고 안전하게 |
| `QUALIFY row_number() OVER (...) = 1` | dedup step |
| `PRAGMA threads=N` | 병렬도 — 기본 코어 수 |
| in-memory `duckdb.connect()` | 서버·파일 없음. 변환은 상태가 없다 (상태는 Parquet과 StateStore에만) |
| `con.sql(q).arrow()` | P1 쿼리 모드·테스트 검증용 Arrow 반환 |

```python
def quote_ident(name: str) -> str:
    return '"' + name.replace('"', '""') + '"'

_CAST_TYPES = {  # cast step 타입 화이트리스트 — 임의 문자열을 타입 위치에 두지 않는다
    "bigint": "BIGINT", "integer": "INTEGER", "double": "DOUBLE",
    "varchar": "VARCHAR", "boolean": "BOOLEAN", "date": "DATE",
    "timestamp": "TIMESTAMP", "decimal": "DECIMAL(38, 9)",
}

def normalize_type(t: str) -> str:
    try:
        return _CAST_TYPES[t.lower()]
    except KeyError:
        raise FatalError(f"unsupported cast type: {t!r}") from None
```

```python
def compile_sql(spec: TransformSpec) -> str:
    src = str(spec.input).rstrip("/")
    ctes = [f"s0 AS (SELECT * FROM read_parquet('{src}/**/*.parquet', union_by_name=true))"]
    idx = 0
    if spec.map:
        idx += 1
        cols = ", ".join(f"{expr} AS {quote_ident(new)}" for new, expr in spec.map.items())
        ctes.append(f"s{idx} AS (SELECT {cols} FROM s{idx-1})")
    for step in spec.steps:
        idx += 1
        ctes.append(f"s{idx} AS ({_compile_step(step, f's{idx-1}')})")
    final = _final_select(spec.steps, f"s{idx}")
    return "WITH " + ",\n".join(ctes) + "\n" + final


def _compile_step(step: Step, prev: str) -> str:
    match step:
        case FilterStep(filter=cond):
            return f"SELECT * FROM {prev} WHERE {cond}"
        case RenameStep(rename=m):
            ex = ", ".join(quote_ident(o) for o in m)
            al = ", ".join(f"{quote_ident(o)} AS {quote_ident(n)}" for o, n in m.items())
            return f"SELECT * EXCLUDE ({ex}), {al} FROM {prev}"
        case CastStep(cast=m):
            rep = ", ".join(
                f"CAST({quote_ident(c)} AS {normalize_type(t)}) AS {quote_ident(c)}"
                for c, t in m.items()
            )
            return f"SELECT * REPLACE ({rep}) FROM {prev}"
        case DedupStep(dedup=keys):
            part = ", ".join(quote_ident(k) for k in keys)
            return (f"SELECT * FROM {prev} QUALIFY row_number() "
                    f"OVER (PARTITION BY {part} ORDER BY {part}) = 1")
        case DeriveStep(derive=m):
            dv = ", ".join(f"{expr} AS {quote_ident(n)}" for n, expr in m.items())
            return f"SELECT *, {dv} FROM {prev}"
        case SelectStep():
            return f"SELECT * FROM {prev}"  # select는 최종 projection에서 처리
```

`select` step은 CTE가 아니라 최종 SELECT로 — 마지막이 아닌 위치의 select는 그 지점에서 projection(중간 select 허용). 구현 시 `_final_select`가 마지막 select step 유무로 `SELECT cols` / `SELECT *`를 결정.

```python
def run_transform(spec: TransformSpec) -> Path:
    sql = compile_sql(spec)
    con = duckdb.connect()                       # in-memory, 상태 없음
    con.execute(f"PRAGMA threads={os.cpu_count()}")
    tmp = spec.output.with_name(spec.output.name + ".tmp")
    tmp.mkdir(parents=True, exist_ok=True)
    try:
        con.execute(
            f"COPY ({sql}) TO '{tmp / 'part-0.parquet'}' "
            "(FORMAT PARQUET, COMPRESSION ZSTD)"
        )
    except duckdb.Error as e:
        raise FatalError(f"transform failed: {e}\n--- compiled SQL ---\n{sql}") from e
    finally:
        con.close()
    if spec.output.exists():
        shutil.rmtree(spec.output)               # 변환 출력은 전체 재계산 의미론 (P0)
    os.replace(tmp, spec.output)
    return spec.output
```

에러에 컴파일된 SQL을 포함한다 — 투명성 원칙(디버깅 시간 = 비용). P1 증분 변환이 들어오면 tmp-swap 부분이 unit 단위 append로 확장된다.

```python
def query(sql: str) -> pa.Table:
    con = duckdb.connect()                      # in-memory, 상태 없음
    try:
        return con.sql(sql).arrow()
    except duckdb.Error as e:
        raise FatalError(str(e)) from e
    finally:
        con.close()
```

CLI: `gladius query "SELECT ..." [--format table|csv|jsonl] [--limit N]`. table 포맷은 DuckDB의 `.show()` 스타일 정렬 출력(직접 구현하지 말고 `con.sql(sql).show()` 활용 검토), csv/jsonl은 pyarrow로 stdout에 스트리밍.
